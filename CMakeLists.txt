CMAKE_MINIMUM_REQUIRED(VERSION 3.30)

SET(PROJECT_NAME valheim)
SET(CMAKE_EXPORT_COMPILE_COMMANDS ON)
FILE(TO_CMAKE_PATH "$ENV{VCPKG}" VCPKG)
SET(CMAKE_TOOLCHAIN_FILE ${VCPKG}/scripts/buildsystems/vcpkg.cmake)

SET(CMAKE_RUNTIME_OUTPUT_DIRECTORY ../)
SET(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ../)
SET(CMAKE_LIBRARY_OUTPUT_DIRECTORY ../)

PROJECT(${PROJECT_NAME} C CXX ASM_NASM)

ADD_SUBDIRECTORY(valheim.core)
ADD_SUBDIRECTORY(valheim.vulkan)
ADD_SUBDIRECTORY(valheim.opengl)
ADD_SUBDIRECTORY(valheim.app)
ADD_SUBDIRECTORY(valheim.app.opengl)
